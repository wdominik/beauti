# Beauti OBS filter: engine, plugin module, bench worker, Catch2 tests, and the
# libobs fixture runners. The presets in CMakePresets.json are the interface
# (docs/BUILD.md); dependencies come from cmake/Dependencies.cmake (buildspec.json).
cmake_minimum_required(VERSION 3.25...4.4)
if(CMAKE_CURRENT_BINARY_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  message(FATAL_ERROR "In-source builds are not supported. Use a preset: cmake --workflow --preset macos")
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/VERSION" BEAUTI_VERSION LIMIT_COUNT 1)
project(Beauti VERSION ${BEAUTI_VERSION} LANGUAGES CXX)
include(CTest)
option(BEAUTI_BUILD_REFERENCE "Build the CPU reference effects library" OFF)
if(BEAUTI_BUILD_REFERENCE)
  add_subdirectory(src/effects)
endif()
option(BEAUTI_BUILD_PLUGIN "Build the OBS module" ON)
option(BEAUTI_ENABLE_TENSORRT "Expose NVIDIA TensorRT RTX backends" ${WIN32})
include(cmake/Dependencies.cmake)
set(BEAUTI_ORT_LIBRARY "" CACHE FILEPATH "ONNX Runtime library (dylib on Mac, import lib on Windows)")
set(BEAUTI_ORT_RUNTIME "" CACHE FILEPATH "ONNX Runtime DLL on Windows")
if(NOT EXISTS "${BEAUTI_ORT_LIBRARY}")
  message(FATAL_ERROR "BEAUTI_ORT_LIBRARY does not exist: ${BEAUTI_ORT_LIBRARY}. See docs/BUILD.md.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_OBJCXX_STANDARD 20)
set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
set(CMAKE_OBJCXX_EXTENSIONS OFF)
set(CMAKE_CXX_VISIBILITY_PRESET hidden)
set(CMAKE_OBJCXX_VISIBILITY_PRESET hidden)
set(CMAKE_VISIBILITY_INLINES_HIDDEN ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
# Sanitizer builds (presets macos-asan, macos-tsan) instrument every target so the
# test executables, the libobs fixture runner, and the test-only module loaded by it
# share one runtime. Never package a sanitized bundle.
set(BEAUTI_SANITIZER "" CACHE STRING "Instrument all targets: address (ASan+UBSan) or thread (TSan)")
set_property(CACHE BEAUTI_SANITIZER PROPERTY STRINGS "" address thread)
if(BEAUTI_SANITIZER)
  if(MSVC)
    message(FATAL_ERROR "BEAUTI_SANITIZER is implemented for Clang and GCC only")
  elseif(BEAUTI_SANITIZER STREQUAL "address")
    set(sanitizer_flags -fsanitize=address,undefined -fno-sanitize-recover=undefined)
  elseif(BEAUTI_SANITIZER STREQUAL "thread")
    set(sanitizer_flags -fsanitize=thread)
  else()
    message(FATAL_ERROR "BEAUTI_SANITIZER must be empty, address, or thread (got ${BEAUTI_SANITIZER})")
  endif()
  # C and C++ only: the Swift adapter stays uninstrumented, as the system libraries are.
  add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX,OBJCXX>:${sanitizer_flags};-fno-omit-frame-pointer>")
  add_link_options("$<$<LINK_LANGUAGE:C,CXX,OBJCXX>:${sanitizer_flags}>")
endif()
add_library(beauti-options INTERFACE)
if(MSVC)
  # /guard:cf and /sdl: control-flow guard and the SDL checks; both compile and link.
  target_compile_options(beauti-options INTERFACE /W4 /EHsc /utf-8 /fp:precise /guard:cf /sdl)
  target_link_options(beauti-options INTERFACE /guard:cf)
  target_compile_definitions(beauti-options INTERFACE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
else()
  # Hardening: stack protector always, fortified libc calls in optimized builds.
  target_compile_options(beauti-options INTERFACE -Wall -Wextra -Wpedantic -Wshadow -Wconversion -Wno-sign-conversion -Wold-style-cast -Werror=return-type -Wno-misleading-indentation -fstack-protector-strong -ffp-contract=off "-ffile-prefix-map=${CMAKE_SOURCE_DIR}=." "-ffile-prefix-map=${CMAKE_BINARY_DIR}=build")
  # _FORTIFY_SOURCE and the sanitizers both replace libc calls; the sanitizer wins.
  if(NOT BEAUTI_SANITIZER)
    target_compile_definitions(beauti-options INTERFACE $<$<NOT:$<CONFIG:Debug>>:_FORTIFY_SOURCE=2>)
  endif()
endif()
add_library(beauti-ort SHARED IMPORTED GLOBAL)
if(WIN32)
  if(NOT EXISTS "${BEAUTI_ORT_RUNTIME}")
    message(FATAL_ERROR "Set BEAUTI_ORT_RUNTIME to the matching onnxruntime.dll.")
  endif()
  set_target_properties(beauti-ort PROPERTIES IMPORTED_IMPLIB "${BEAUTI_ORT_LIBRARY}" IMPORTED_LOCATION "${BEAUTI_ORT_RUNTIME}")
else()
  set_target_properties(beauti-ort PROPERTIES IMPORTED_LOCATION "${BEAUTI_ORT_IMPORT}")
endif()
configure_file(cmake/beauti-version.h.in generated/beauti-version.h @ONLY)
find_package(Threads REQUIRED)
add_library(beauti-engine STATIC src/core/eyes.cpp src/core/engine.cpp src/backends/backend.cpp src/core/paths.cpp)
target_include_directories(beauti-engine PUBLIC src/core src/backends "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_include_directories(beauti-engine SYSTEM PUBLIC third_party/onnxruntime/include)
target_compile_definitions(beauti-engine PUBLIC BEAUTI_ENABLE_TENSORRT=$<BOOL:${BEAUTI_ENABLE_TENSORRT}>)
if(WIN32 AND BEAUTI_ENABLE_TENSORRT)
  configure_file(cmake/beauti-tensorrt.h.in generated/beauti-tensorrt.h @ONLY)
endif()
target_link_libraries(beauti-engine PUBLIC beauti-options beauti-ort Threads::Threads)
if(WIN32)
  target_link_libraries(beauti-engine PUBLIC shell32 ole32)
endif()
if(APPLE)
  # Core AI has a Swift interface only. The adapter is a separate library with a C
  # interface (src/backends/coreai.h) that the engine links; see docs/DECISIONS.md.
  enable_language(Swift)
  add_library(beauti-coreai SHARED src/backends/coreai.swift)
  set_target_properties(beauti-coreai PROPERTIES OUTPUT_NAME beauti_coreai INSTALL_NAME_DIR "@rpath")
  target_compile_options(beauti-coreai PRIVATE -parse-as-library "SHELL:-file-prefix-map ${CMAKE_SOURCE_DIR}=.")
  target_sources(beauti-engine PRIVATE src/backends/coreai_model.cpp)
  target_link_libraries(beauti-engine PUBLIC beauti-coreai)
endif()
add_executable(beauti-bench tools/benchmarks/worker.cpp)
target_link_libraries(beauti-bench PRIVATE beauti-engine)
if(APPLE)
  set_target_properties(beauti-bench PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@loader_path/../../Frameworks")
  set(BEAUTI_BINARY_DIR "beauti.plugin/Contents/MacOS")
  set(BEAUTI_DATA_DIR "beauti.plugin/Contents/Resources")
  install(TARGETS beauti-bench RUNTIME DESTINATION "${BEAUTI_DATA_DIR}/bin")
  install(FILES "${BEAUTI_ORT_IMPORT}" DESTINATION "beauti.plugin/Contents/Frameworks" RENAME libbeauti_onnxruntime.dylib)
  install(TARGETS beauti-coreai LIBRARY DESTINATION "beauti.plugin/Contents/Frameworks")
  configure_file(cmake/Info.plist.in generated/Info.plist @ONLY)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/generated/Info.plist" DESTINATION "beauti.plugin/Contents")
elseif(WIN32)
  if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The Windows build requires x64.")
  endif()
  # OBS 33 plugin layout: the module, its runtime DLLs, and the bench tool sit in the
  # plugin folder itself; OBS 32 and earlier expected bin/64bit and are not supported.
  set(BEAUTI_BINARY_DIR "beauti")
  set(BEAUTI_DATA_DIR "beauti/data")
  install(TARGETS beauti-bench RUNTIME DESTINATION "${BEAUTI_BINARY_DIR}")
  install(FILES "${BEAUTI_ORT_RUNTIME}" DESTINATION "${BEAUTI_BINARY_DIR}")
else()
  message(FATAL_ERROR "Package layouts are currently implemented for macOS and Windows only.")
endif()
if(BEAUTI_BUILD_PLUGIN)
  include(cmake/OBS.cmake)
  add_library(beauti MODULE src/plugin/filter.cpp src/plugin/gpu.cpp src/plugin/notice.cpp)
  target_link_libraries(beauti PRIVATE beauti-engine OBS::libobs OBS::obs-frontend-api)
  set_target_properties(beauti PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
  if(APPLE)
    target_link_libraries(beauti PRIVATE "-framework CoreFoundation")
    set_target_properties(beauti PROPERTIES SUFFIX "" BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks;@loader_path/../Frameworks")
  endif()
  install(TARGETS beauti LIBRARY DESTINATION "${BEAUTI_BINARY_DIR}" RUNTIME DESTINATION "${BEAUTI_BINARY_DIR}")
  install(DIRECTORY data/ DESTINATION "${BEAUTI_DATA_DIR}" PATTERN ".DS_Store" EXCLUDE)
endif()
if(BUILD_TESTING)
  # Catch2 is compiled without the project's warning set: it is third-party code.
  add_library(beauti-catch2 STATIC "${BEAUTI_CATCH2_DIR}/catch_amalgamated.cpp")
  target_include_directories(beauti-catch2 SYSTEM PUBLIC "${BEAUTI_CATCH2_DIR}")
  set_target_properties(beauti-catch2 PROPERTIES COMPILE_WARNING_AS_ERROR OFF)
  add_executable(beauti-tests tests/native/eyes.cpp tests/native/continuity.cpp tests/native/pixel_quantize.cpp tests/native/provider_policy.cpp tests/native/provider_lifetime.cpp tests/native/pacing.cpp tests/native/ramp_scene_stats.cpp tests/native/face_tracker.cpp)
  target_link_libraries(beauti-tests PRIVATE beauti-engine beauti-catch2)
  # One CTest entry per tag keeps the names stable: ctest -R continuity.
  foreach(tag IN ITEMS continuity pixel-quantize provider-policy eyes pacing ramp scene-cut live-stats face-tracker)
    add_test(NAME ${tag} COMMAND beauti-tests "[${tag}]")
  endforeach()
  if(WIN32)
    add_custom_command(TARGET beauti-tests POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different "${BEAUTI_ORT_RUNTIME}" "$<TARGET_FILE_DIR:beauti-tests>")
  endif()
endif()
option(BEAUTI_BUILD_OBS_TESTS "Build the separate libobs fixture runner" ON)
if(BEAUTI_BUILD_PLUGIN AND BEAUTI_BUILD_OBS_TESTS)
  if(BUILD_TESTING)
    add_library(beauti-continuity-test MODULE src/plugin/filter.cpp src/plugin/gpu.cpp src/plugin/notice.cpp)
    target_compile_definitions(beauti-continuity-test PRIVATE BEAUTI_CONTINUITY_TEST=1)
    target_link_libraries(beauti-continuity-test PRIVATE beauti-engine OBS::libobs OBS::obs-frontend-api)
    set_target_properties(beauti-continuity-test PROPERTIES PREFIX "")
    if(APPLE)
      target_link_libraries(beauti-continuity-test PRIVATE "-framework CoreFoundation")
    endif()
  endif()
  if(APPLE)
    enable_language(OBJCXX)
    set_source_files_properties(tests/obs/obs_headless.cpp PROPERTIES LANGUAGE OBJCXX)
  endif()
  add_executable(beauti-obs-check tests/obs/obs_headless.cpp)
  add_executable(beauti-gpu-check tests/native/gpu_pipeline.cpp src/plugin/gpu.cpp)
  target_include_directories(beauti-gpu-check PRIVATE src/plugin)
  target_link_libraries(beauti-gpu-check PRIVATE beauti-engine OBS::libobs)
  target_include_directories(beauti-obs-check PRIVATE src/core "${CMAKE_CURRENT_BINARY_DIR}/generated")
  target_link_libraries(beauti-obs-check PRIVATE beauti-options OBS::libobs Threads::Threads)
  if(APPLE)
    target_link_libraries(beauti-obs-check PRIVATE "-framework Cocoa")
    set_source_files_properties(tests/native/gpu_pipeline.cpp PROPERTIES LANGUAGE OBJCXX)
    target_link_libraries(beauti-gpu-check PRIVATE "-framework Cocoa")
  endif()
endif()
